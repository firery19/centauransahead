

defined_text = { # when u do nothing
	name = LepusRebellionProgress

		}

				original_tag = HYS

			}
			check_variable = { lepus_revolution_counter < 15 }

		}
		localization_key = lepus_low_revolution
	}

		trigger = {
			OR = {
				original_tag = HYS

			}
			check_variable = { lepus_revolution_counter < 25 }

		}
		localization_key = lepus_medium_revolution
	}
	text = {
		trigger = {
			OR = {
				original_tag = HYS
				original_tag = URS
			}
			check_variable = { lepus_revolution_counter < 35 }

		}
		localization_key = lepus_high_revolution
	}
	text = {
		trigger = {
			OR = {
				original_tag = HYS
				original_tag = URS
			}
			check_variable = {

				value = 35

			}
		}
		localization_key = lepus_total_revolution
	}
}
