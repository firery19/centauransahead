country_event = {
	id = formageneric.1 # Cetan units flee into Aquila!
	title = formageneric.1.t
	desc = formageneric.1.d
	picture = GFX_report_event_generic_read_write
	
	is_triggered_only = yes
	
	option = {
		name = formageneric.1.a
		add_manpower = 20000
		add_political_power = -50
		if = {
			limit = {
			CET = { has_tech = infantry_equipment_2 }
			}
			add_equipment_to_stockpile = {
				type = infantry_equipment_2
				amount = 600
				producer = CET
			}
			add_equipment_to_stockpile = {
				type = infantry_equipment_1
				amount = 1200
				producer = CET
			}
		}
		else_if = {
			limit = {
			CET = { has_tech = infantry_equipment_1 }
			}
			add_equipment_to_stockpile = {
				type = infantry_equipment_1
				amount = 1000
				producer = CET
			}
			add_equipment_to_stockpile = {
				type = infantry_equipment_0
				amount = 800
				producer = CET
			}
		}
		else = {
			add_equipment_to_stockpile = {
				type = infantry_equipment_0
				amount = 1800
				producer = CET
			}
		}
	}
	option = {
		name = formageneric.1.b # turn them back>?? what??
		add_political_power = 10
		add_war_support = 0.1
	}
}
